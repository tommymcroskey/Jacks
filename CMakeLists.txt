cmake_minimum_required(VERSION 3.20)
project(Jacks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(jacks_valuation STATIC src/valuation.cpp)
target_include_directories(jacks_valuation PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_library(jacks_deposit STATIC src/deposit.cpp)
target_link_libraries(jacks_deposit PUBLIC jacks_valuation)
add_library(jacks_replay STATIC src/replay.cpp)
target_link_libraries(jacks_replay PUBLIC jacks_deposit)

include(CTest)
if(BUILD_TESTING)
  include(FetchContent)
  # Official GoogleTest v1.14.0: immutable commit verified with git ls-remote.
  FetchContent_Declare(googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG f8d7d77c06936315286eb55f8de22cd23c188571)
  set(gtest_force_shared_crt ON CACHE BOOL "Match MSVC runtime" FORCE)
  set(BUILD_GMOCK OFF CACHE BOOL "Only GoogleTest needed" FORCE)
  set(INSTALL_GTEST OFF CACHE BOOL "No global installation" FORCE)
  FetchContent_MakeAvailable(googletest)
  add_executable(jacks_tests tests/infrastructure_test.cpp tests/valuation_test.cpp tests/deposit_test.cpp)
  target_link_libraries(jacks_tests PRIVATE jacks_deposit GTest::gtest_main)
  include(GoogleTest)
  gtest_discover_tests(jacks_tests DISCOVERY_MODE PRE_TEST)
  add_executable(jacks_replay_tests tests/replay_test.cpp)
  target_link_libraries(jacks_replay_tests PRIVATE jacks_replay GTest::gtest_main)
  gtest_discover_tests(jacks_replay_tests DISCOVERY_MODE PRE_TEST)
endif()
# Apply warnings to our code, not the downloaded dependency.
foreach(jacks_target jacks_valuation jacks_deposit jacks_replay jacks_tests jacks_replay_tests)
  if(TARGET ${jacks_target})
    if(MSVC)
      target_compile_options(${jacks_target} PRIVATE /W4 /WX /permissive- /Zc:__cplusplus)
    else()
      target_compile_options(${jacks_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
  endif()
endforeach()
